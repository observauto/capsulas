with open('fullCapsules.ts', 'r', encoding='utf-8') as f:

def replace_quiz(match):
    question = match.group(1)
    options = match.group(2)
    correct_index = match.group(3)
    explanation = match.group(4)
    
    # Generar ID único basado en un hash de la pregunta
    quiz_id = f"quiz_{hash(question) % 100000}"

        id: "{quiz_id}",
        order: 1,
        question: "{question}",
        options: [{options}],
        correctIndex: {correct_index},
        explanation: "{explanation}"
      }}
    ]'''

# Reemplazar con lookbehind para capturar correctamente
content_fixed = re.sub(pattern, replace_quiz, content, flags=re.DOTALL)

with open('fullCapsules.ts', 'w', encoding='utf-8') as f:
    f.write(content_fixed)

print("Quiz objects converted to arrays successfully!")

import re

# Leer el archivo
with open('src/data/fullCapsules.ts', 'r', encoding='utf-8') as f:
    lines = f.readlines()

output = []
i = 0
quiz_counter = 1

while i < len(lines):
    line = lines[i]
    
    # Detectar inicio de quiz como objeto
    if '    quiz: {' in line and 'question:' in ''.join(lines[i:i+15]):
        # Capturar el bloque quiz
        quiz_lines = [line]
        i += 1
        indent_count = 0
        
        while i < len(lines):
            quiz_lines.append(lines[i])
            if '{' in lines[i]:
                indent_count += 1
            if '}' in lines[i]:
                if indent_count == 0:
                    break
                indent_count -= 1
            i += 1
        
        # Extraer datos del quiz
        quiz_text = ''.join(quiz_lines)
        
        # Buscar question, options, correctIndex, explanation
        question_match = re.search(r'question: "(.*?)"', quiz_text, re.DOTALL)
        options_match = re.search(r'options: \[(.*?)\]', quiz_text, re.DOTALL)
        index_match = re.search(r'correctIndex: (\d+)', quiz_text)
        explanation_match = re.search(r'explanation: "(.*?)"', quiz_text, re.DOTALL)
        
        if all([question_match, options_match, index_match, explanation_match]):
            question = question_match.group(1)
            options = options_match.group(1)
            correct_idx = index_match.group(1)
            explanation = explanation_match.group(1)
            
            # Generar nuevo formato
            new_quiz = f'''    quiz: [
      {{
        id: "quiz_{quiz_counter}",
        order: 1,
        question: "{question}",
        options: [{options}],
        correctIndex: {correct_idx},
        explanation: "{explanation}"
      }}
    ],
'''
            output.append(new_quiz)
            quiz_counter += 1
        else:
            # Si no se pudo parsear, mantener original
            output.extend(quiz_lines)
    else:
        output.append(line)
    
    i += 1

# Escribir resultado
with open('src/data/fullCapsules.ts', 'w', encoding='utf-8') as f:
    f.writelines(output)
